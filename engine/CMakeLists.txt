cmake_minimum_required(VERSION 3.16)
project(apex_matching_engine LANGUAGES CXX)
option(ENABLE_TSAN "Build engine and tests with ThreadSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(apex_core src/order_book.cpp src/matching_engine.cpp src/protocol.cpp)
target_include_directories(apex_core PUBLIC include)
target_compile_options(apex_core PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
target_link_libraries(apex_core PUBLIC Threads::Threads)
add_executable(apex-engine src/main.cpp)
target_link_libraries(apex-engine PRIVATE apex_core)
add_executable(engine-tests tests/test_engine.cpp)
target_link_libraries(engine-tests PRIVATE apex_core)
if(ENABLE_TSAN)
  foreach(target apex_core apex-engine engine-tests)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=thread)
  endforeach()
endif()
enable_testing()
add_test(NAME engine-tests COMMAND engine-tests)
